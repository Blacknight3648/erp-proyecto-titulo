/**
 * Clase base para entidades JPA que necesitan auditorÃ­a temporal automÃ¡tica.
 *
 * Al extenderla, la entidad gana las columnas `created_at` y `updated_at`
 * que Spring rellena automÃ¡ticamente en cada save(), sin necesidad de tocar
 * el cÃ³digo de los UseCase o mappers.
 *
 * Requiere @EnableJpaAuditing en la configuraciÃ³n (ver JpaAuditingConfig).
 */
@MappedSuperclass
@EntityListeners(AuditingEntityListener.class)
@Getter
@Setter
public abstract class AuditableJpaEntity {

    @CreatedDate
    @Column(name = "created_at", updatable = false)
    private LocalDateTime createdAt;

    @LastModifiedDate
    @Column(name = "updated_at")
    private LocalDateTime updatedAt;
}

import jakarta.persistence.Column;
import jakarta.persistence.MappedSuperclass;
import jakarta.persistence.PrePersist;
import jakarta.persistence.PreUpdate;
import lombok.Data;
import java.time.LocalDateTime;

/**
 * Base abstract class for JPA entities to include common auditing fields.
 * Relocated to infrastructure layer as part of DDD refactoring.
 */
@MappedSuperclass
@Data
public abstract class BaseEntity {

    @Column(name = "creado_en", nullable = false, updatable = false)
    private LocalDateTime creadoEn;

    @Column(name = "actualizado_en", nullable = false)
    private LocalDateTime actualizadoEn;

    @Column(name = "activo", nullable = false)
    private Boolean activo = true;

    @PrePersist
    protected void onCreate() {
        creadoEn = LocalDateTime.now();
        actualizadoEn = LocalDateTime.now();
    }

    @PreUpdate
    protected void onUpdate() {
        actualizadoEn = LocalDateTime.now();
    }
}
